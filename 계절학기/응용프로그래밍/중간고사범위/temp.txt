#include <math.h>
#include <stdio.h>
#include <stdlib.h>

void eratos(int number, char array[]);
void print_primes(int count, char *array);

int main() {
    int number;
    char *array;

    printf("정수를 입력하세요: ");
    scanf("%d", &number);
    if (number > 2) {
        array = (char *)calloc(number + 1, sizeof(char));
        if (array != NULL) {
            printf("%d개의 배열을 할당하였습니다.\n", number + 1);
            eratos(number, array);
            print_primes(number, array);
            free(array);
        }
    }
    return 0;
}

void eratos(int number, char array[]) {
    int n, m, to, last, index;
    to = (int)sqrt((double)number);
    for (n = 2; n <= to; n++) {
        last = number / n;
        for (m = 2; m <= last; m++) {
            index = n * m;
            array[index] = 1;
        }
    }
}

void print_primes(int count, char *array) {
    int n;
    printf("%d의 소수:\n", count);
    for (n = 2; n <= count; n++) {
        if (array[n] == 0) {
            printf("%9d", n);
        }
    }
    printf("\n");
}
